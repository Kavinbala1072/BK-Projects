CREATE PROCEDURE sp_GetWeaveTypeList
    @Company_No BIGINT,
    @SearchText NVARCHAR(100) = NULL
AS
BEGIN
    SELECT * FROM WeaveType_Table 
    WHERE Company_No = @Company_No 
    AND (@SearchText IS NULL OR Weave_Name LIKE '%' + @SearchText + '%')
    ORDER BY Weave_Name ASC;
END
GO
