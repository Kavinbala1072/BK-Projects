CREATE PROCEDURE sp_GetLedgerGroupList
    @Company_No BIGINT,
    @SearchText NVARCHAR(100) = NULL
AS
BEGIN
    SET NOCOUNT ON;
    SELECT g.LedgerGroup_Sno, g.LedgerGroup_Name, g.IsActive,
           CASE g.Nature 
                WHEN 0 THEN 'Assets' 
                WHEN 1 THEN 'Liabilities' 
                WHEN 2 THEN 'Income' 
                WHEN 3 THEN 'Expenses' 
                ELSE 'Other' END as NatureText, 
           ISNULL(p.LedgerGroup_Name, 'PRIMARY') as ParentName 
    FROM LedgerGroup_Table g 
    LEFT JOIN LedgerGroup_Table p ON g.Ledgergroup_Under = p.LedgerGroup_Sno
    WHERE g.Company_No = @Company_No
      AND (@SearchText IS NULL OR g.LedgerGroup_Name LIKE '%' + @SearchText + '%')
    ORDER BY g.LedgerGroup_Sno DESC;
END
