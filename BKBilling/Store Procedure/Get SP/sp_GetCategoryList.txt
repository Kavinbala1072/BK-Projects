CREATE PROCEDURE sp_GetCategoryList
    @Company_No BIGINT,
    @SearchText NVARCHAR(100) = NULL
AS
BEGIN
    SELECT 
        c.ItemGroup_Sno, 
        c.ItemGroup_Name, 
        c.IsActive,
        ISNULL(p.ItemGroup_Name, 'PRIMARY') as ParentCategoryName
    FROM ItemGroup_Table c
    LEFT JOIN ItemGroup_Table p ON c.ItemGroup_Under = p.ItemGroup_Sno
    WHERE c.Company_No = @Company_No 
    AND (@SearchText IS NULL OR c.ItemGroup_Name LIKE '%' + @SearchText + '%')
    ORDER BY c.ItemGroup_Name ASC;
END
GO
