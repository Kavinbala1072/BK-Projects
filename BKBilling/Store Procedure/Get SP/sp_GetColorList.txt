CREATE PROCEDURE sp_GetColorList
    @Company_No BIGINT,
    @SearchText NVARCHAR(100) = NULL
AS
BEGIN
    SELECT * FROM Color_Table 
    WHERE Company_No = @Company_No 
    AND (@SearchText IS NULL OR Color_Name LIKE '%' + @SearchText + '%')
    ORDER BY Color_Name ASC;
END
GO
