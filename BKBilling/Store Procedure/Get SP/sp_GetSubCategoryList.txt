CREATE PROCEDURE sp_GetSubCategoryList
    @Company_No BIGINT,
    @SearchText NVARCHAR(100) = NULL
AS
BEGIN
    SELECT sc.*, c.ItemGroup_Name as MainCategoryName
    FROM ItemSubCategory_Table sc
    INNER JOIN ItemGroup_Table c ON sc.Category_No = c.ItemGroup_Sno
    WHERE sc.Company_No = @Company_No 
    AND (@SearchText IS NULL OR sc.SubCat_Name LIKE '%' + @SearchText + '%' OR c.ItemGroup_Name LIKE '%' + @SearchText + '%')
    ORDER BY c.ItemGroup_Name, sc.SubCat_Name ASC;
END
