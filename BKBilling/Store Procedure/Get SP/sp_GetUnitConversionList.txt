/*                                  
    Description       : Fetching all Unit Conversions for the GridView.
    Tables            : UnitConversion_Table, Unit_Table
    Generated By      : Kavin
    Generated ON      : 08/07/2026
    Version           : 1
*/
CREATE PROCEDURE sp_GetUnitConversionList
    @Company_No BIGINT
AS
BEGIN
    SET NOCOUNT ON;
    SELECT C.Conv_Sno, U1.Unit_Sname as MainUnit, 
           C.Multiplier, U2.Unit_Sname as SubUnit, C.IsActive
    FROM UnitConversion_Table C
    JOIN Unit_Table U1 ON C.MainUnit_Sno = U1.Unit_Sno
    JOIN Unit_Table U2 ON C.SubUnit_Sno = U2.Unit_Sno
    WHERE C.Company_No = @Company_No 
    ORDER BY C.Conv_Sno DESC;
END
GO
