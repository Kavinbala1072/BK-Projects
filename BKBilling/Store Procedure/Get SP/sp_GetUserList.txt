CREATE PROCEDURE sp_GetUserList
    @Company_No BIGINT
AS
BEGIN
    SET NOCOUNT ON;
    SELECT User_Sno, Username, FullName, Role, Phone, IsActive, Join_Date 
    FROM User_Table 
    WHERE Company_No = @Company_No AND Role <> 'Admin' 
    ORDER BY User_Sno DESC;
END
GO
