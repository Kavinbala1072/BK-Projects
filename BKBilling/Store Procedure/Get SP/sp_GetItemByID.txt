/*                                  
    Description       : Fetching a single item for Edit mode.
    Tables            : Item_Table
    Generated By      : Kavin
*/
CREATE PROCEDURE sp_GetItemByID
    @Item_Sno INT
AS
BEGIN
    SET NOCOUNT ON;
    SELECT 
        I.*, 
        ISNULL(S.Stk_Value, 0) as OpeningQty 
    FROM Item_Table I
    LEFT JOIN ClosingStk_table S ON I.Item_Sno = S.Item_no
    WHERE I.Item_Sno = @Item_Sno;
END
GO
