/*                                  
    Description       : Retrieves the list of Areas for a specific company with Parent Area names and search filtering.
    Tables            : Area_table
    Generated By      : Kavin
    Generated ON      : 07/07/2026
    Version           : 1
*/
CREATE PROCEDURE sp_GetAreaList
    @Company_No BIGINT,
    @SearchText NVARCHAR(100) = NULL
AS
BEGIN
    SET NOCOUNT ON;

    SELECT a.Area_Sno, a.Area_Name, a.IsActive,
           ISNULL(p.Area_Name, 'PRIMARY') as ParentAreaName 
    FROM Area_table a
    LEFT JOIN Area_table p ON a.Area_Under = p.Area_Sno
    WHERE a.Company_No = @Company_No
      AND (@SearchText IS NULL OR a.Area_Name LIKE '%' + @SearchText + '%')
    ORDER BY a.Area_Sno DESC;
END
