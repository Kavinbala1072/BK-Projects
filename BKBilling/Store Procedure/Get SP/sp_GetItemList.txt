/*                                  
    Description       : Fetching full item list with joined names for the Directory.
    Tables            : Item_Table, ItemGroup_Table, Brand_Table, Unit_Table
    Generated By      : Kavin
*/

CREATE PROCEDURE sp_GetItemList
    @Company_No BIGINT,
    @SearchText NVARCHAR(100) = NULL
AS
BEGIN
    SET NOCOUNT ON;
    SELECT 
        I.*, 
        G.ItemGroup_Name,
        SC.SubCat_Name,
        C.Color_Name,
        W.Weave_Name,
        U.Unit_Sname,
        ISNULL(S.Stk_Value, 0) as OpeningQty
    FROM Item_Table I 
    LEFT JOIN ItemGroup_Table G ON I.ItemGroup_No = G.ItemGroup_Sno
    LEFT JOIN ItemSubCategory_Table SC ON I.SubCategory_Sno = SC.SubCat_Sno
    LEFT JOIN Color_Table C ON I.Color_Sno = C.Color_Sno
    LEFT JOIN WeaveType_Table W ON I.Weave_Sno = W.Weave_Sno
    LEFT JOIN Unit_Table U ON I.ItemUnit_No = U.Unit_Sno
    LEFT JOIN ClosingStk_table S ON I.Item_Sno = S.Item_no AND I.Company_No = S.Company_No
    WHERE I.Company_No = @Company_No 
      AND (@SearchText IS NULL OR I.Item_Name LIKE '%' + @SearchText + '%' OR I.Item_Code LIKE '%' + @SearchText + '%')
    ORDER BY I.Item_Sno DESC;
END
GO
