CREATE PROCEDURE sp_GetSupplierList
    @Company_No BIGINT,
    @LedgerGroup_no INT = 1000000033,
    @SearchText NVARCHAR(100) = NULL
AS
BEGIN
    SET NOCOUNT ON;
    SELECT L.*, ISNULL(A.Area_Name, 'PRIMARY') AS Area_Name 
    FROM Ledger_Table L 
    LEFT JOIN Area_table A ON L.Area_no = A.Area_Sno
    WHERE L.Company_No = @Company_No 
      AND L.LedgerGroup_no = @LedgerGroup_no
      AND (@SearchText IS NULL OR L.ledger_name LIKE '%' + @SearchText + '%' OR L.Ledger_Phone LIKE '%' + @SearchText + '%')
    ORDER BY L.ledger_name ASC;
END
GO
