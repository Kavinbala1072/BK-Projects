CREATE PROCEDURE sp_GetAreaDropdown
    @Company_No BIGINT
AS
BEGIN
    SET NOCOUNT ON;

    SELECT 
        Area_Sno, 
        Area_Name 
    FROM Area_table 
    WHERE Company_No = @Company_No 
      AND IsActive = 1 
    ORDER BY Area_Name ASC;
END
