CREATE PROCEDURE sp_GetNextVoucherNumber
    @Company_No BIGINT,
    @Voucher_Name NVARCHAR(50)
AS
BEGIN
    DECLARE @Prefix NVARCHAR(20), @Suffix NVARCHAR(20), @NextNo INT, @Width INT
    
    SELECT 
        @Prefix = Prefix, 
        @Suffix = Suffix, 
        @NextNo = Current_No + 1, 
        @Width = Padding_Width 
    FROM VoucherType_Table 
    WHERE Company_No = @Company_No AND Voucher_Name = @Voucher_Name AND IsActive = 1

    -- Format: Prefix + 00001 + Suffix
    SELECT @Prefix + RIGHT('00000000' + CAST(@NextNo AS VARCHAR), @Width) + @Suffix AS Generated_No
END
