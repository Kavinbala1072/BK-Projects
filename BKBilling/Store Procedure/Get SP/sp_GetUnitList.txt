CREATE PROCEDURE sp_GetUnitList
    @Company_No BIGINT,
    @SearchText NVARCHAR(100) = NULL
AS
BEGIN
    SET NOCOUNT ON;
    SELECT * FROM Unit_Table 
    WHERE Company_No = @Company_No
      AND (@SearchText IS NULL OR Unit_Name LIKE '%' + @SearchText + '%' OR Unit_Sname LIKE '%' + @SearchText + '%')
    ORDER BY Unit_Sno DESC;
END
GO
