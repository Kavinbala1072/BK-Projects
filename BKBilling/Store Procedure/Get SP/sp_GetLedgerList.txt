/*                                  
    Description       : Retrieves the list of all ledgers for a specific company, including 
                        associated Group names. Supports searching by Ledger Name or Code.
    Tables            : Ledger_Table, LedgerGroup_Table
    Generated By      : Kavin
    Generated ON      : 07/07/2026
    Version           : 1
*/
CREATE PROCEDURE sp_GetLedgerList
    @Company_No BIGINT,
    @SearchText NVARCHAR(100) = NULL
AS
BEGIN
    SET NOCOUNT ON;
    SELECT L.*, G.LedgerGroup_Name 
    FROM Ledger_Table L
    LEFT JOIN LedgerGroup_Table G ON L.LedgerGroup_no = G.LedgerGroup_Sno
    WHERE L.Company_No = @Company_No
      AND (@SearchText IS NULL OR L.ledger_name LIKE '%' + @SearchText + '%' OR L.ledger_code LIKE '%' + @SearchText + '%')
    ORDER BY L.Ledger_Sno DESC;
END
GO
