CREATE PROCEDURE sp_SaveLedgerGroup
    @Action NVARCHAR(10),
    @LedgerGroup_Sno INT = 0,
    @Company_No BIGINT,
    @LedgerGroup_Name NVARCHAR(100),
    @Ledgergroup_Under INT,
    @Nature NVARCHAR(50),
    @IsActive BIT
AS
BEGIN
    SET NOCOUNT ON;

    -- Duplicate Check
    IF EXISTS (SELECT 1 FROM LedgerGroup_Table WHERE Company_No = @Company_No AND LedgerGroup_Name = @LedgerGroup_Name AND LedgerGroup_Sno <> @LedgerGroup_Sno)
    BEGIN
        RAISERROR('This Group Name already exists.', 16, 1);
        RETURN;
    END

    IF @Action = 'INSERT'
    BEGIN
        INSERT INTO LedgerGroup_Table (Company_No, LedgerGroup_Name, Ledgergroup_Under, Nature, IsActive, Created_Date)
        VALUES (@Company_No, @LedgerGroup_Name, @Ledgergroup_Under, @Nature, @IsActive, GETDATE());
    END
    ELSE IF @Action = 'UPDATE'
    BEGIN
        UPDATE LedgerGroup_Table SET 
            LedgerGroup_Name = @LedgerGroup_Name, 
            Ledgergroup_Under = @Ledgergroup_Under, 
            Nature = @Nature, 
            IsActive = @IsActive 
        WHERE LedgerGroup_Sno = @LedgerGroup_Sno AND Company_No = @Company_No;
    END
END
