/*                                  
    Description       : Saving (Insert/Update) Unit details with Duplicate validation.
    Tables            : Unit_Table
    Generated By      : Kavin
    Generated ON      : 07/07/2026
    Version           : 1
*/
CREATE PROCEDURE sp_SaveUnit
    @Action         NVARCHAR(10),
    @Unit_Sno       INT = 0,
    @Company_No     BIGINT,
    @Unit_Name      NVARCHAR(50),
    @Unit_Sname     NVARCHAR(20),
    @Decimal_Places INT,
    @IsActive       BIT
AS
BEGIN
    SET NOCOUNT ON;

    -- Duplicate Check: Check if Unit Name or Symbol exists in the same company
    IF EXISTS (SELECT 1 FROM Unit_Table WHERE Company_No = @Company_No AND (Unit_Name = @Unit_Name OR Unit_Sname = @Unit_Sname) AND Unit_Sno <> @Unit_Sno)
    BEGIN
        RAISERROR('This Unit Name or Symbol already exists.', 16, 1);
        RETURN;
    END

    IF @Action = 'INSERT'
    BEGIN
        INSERT INTO Unit_Table (Company_No, Unit_Name, Unit_Sname, Decimal_Places, IsActive, Created_Date)
        VALUES (@Company_No, @Unit_Name, @Unit_Sname, @Decimal_Places, @IsActive, GETDATE());
    END
    ELSE IF @Action = 'UPDATE'
    BEGIN
        UPDATE Unit_Table SET 
            Unit_Name = @Unit_Name, 
            Unit_Sname = @Unit_Sname, 
            Decimal_Places = @Decimal_Places, 
            IsActive = @IsActive 
        WHERE Unit_Sno = @Unit_Sno AND Company_No = @Company_No;
    END
END
GO
