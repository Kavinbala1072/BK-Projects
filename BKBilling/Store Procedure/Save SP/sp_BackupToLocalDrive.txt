/*                                  
    Description       : Backs up the DB to a physical local drive folder.
    Generated By      : Kavin
    Version           : 1
*/
CREATE PROCEDURE sp_BackupToLocalDrive
    @FolderPath NVARCHAR(500)
AS
BEGIN
    SET NOCOUNT ON;
    DECLARE @DbName NVARCHAR(100) = DB_NAME();
    DECLARE @FileName NVARCHAR(500);
    
    SET @FileName = @FolderPath + @DbName + '_Full_' + 
                    REPLACE(REPLACE(REPLACE(CONVERT(NVARCHAR, GETDATE(), 120), '-', ''), ' ', '_'), ':', '') + '.bak';

    BACKUP DATABASE @DbName 
    TO DISK = @FileName 
    WITH FORMAT, MEDIANAME = 'LocalDriveBackup', NAME = 'Full Backup';
    
    -- Return the filename to C# so we can show it in the alert
    SELECT @FileName AS FinalPath;
END
GO
