CREATE OR ALTER PROCEDURE sp_SaveItem
    @Action          NVARCHAR(10),
    @Item_Sno        INT = 0,
    @Company_No      BIGINT,
    @User_No         INT,
    @Item_Type       NVARCHAR(20),
    @Item_Name       NVARCHAR(200),
    @Item_Code       NVARCHAR(50),
    @HSN_Code        NVARCHAR(20),
    @ItemGroup_No    INT, 
    @SubCategory_Sno INT,
    @Brand_Sno       INT,
    @Color_SNo       INT,
    @Weave_Sno       INT,
    @ItemUnit_No     INT,
    @AltUnit_No      INT,
    @Conv_Factor     DECIMAL(18, 4),
    @GST_Sno         INT,
    @Purchase_Rate   DECIMAL(18, 2),
    @Selling_Price   DECIMAL(18, 2),
    @IsActive        BIT,
    @OpeningQty      DECIMAL(18, 2),
    @Barcode         NVARCHAR(50),
    @Item_Image      NVARCHAR(MAX),
    @Min_Stock       DECIMAL(18,2),
    @Max_Stock       DECIMAL(18,2),
    @Batch_Enabled   BIT,
    @Serial_Enabled  BIT
AS
BEGIN
    SET NOCOUNT ON;

    IF @Action = 'INSERT'
    BEGIN
        IF EXISTS(SELECT 1 FROM Item_Table WHERE Company_No=@Company_No AND Item_Code=@Item_Code)
        BEGIN
            RAISERROR('Item Code already exists.', 16, 1);
            RETURN;
        END

        INSERT INTO Item_Table (
            Company_No, User_No, Item_Type, Item_Name, Item_Code, HSN_Code, ItemGroup_No, 
            SubCategory_Sno, Brand_Sno, Color_Sno, Weave_Sno, ItemUnit_No, AltUnit_No, 
            Conv_Factor, GST_Sno, Purchase_Rate, Selling_Price, IsActive, Created_Date, 
            Barcode, Item_Image, Min_Stock, Max_Stock, Batch_Enabled, Serial_Enabled
        )
        VALUES (
            @Company_No, @User_No, @Item_Type, @Item_Name, @Item_Code, @HSN_Code, @ItemGroup_No, 
            @SubCategory_Sno, @Brand_Sno, @Color_SNo, @Weave_Sno, @ItemUnit_No, @AltUnit_No, 
            @Conv_Factor, @GST_Sno, @Purchase_Rate, @Selling_Price, @IsActive, GETDATE(), 
            @Barcode, @Item_Image, @Min_Stock, @Max_Stock, @Batch_Enabled, @Serial_Enabled
        );
        
        DECLARE @NewID INT = SCOPE_IDENTITY();
        INSERT INTO ClosingStk_table (Company_No, Item_no, Stk_Value, Stk_date) 
        VALUES (@Company_No, @NewID, @OpeningQty, GETDATE());
    END
    ELSE
    BEGIN
        UPDATE Item_Table SET 
            Item_Type=@Item_Type, Item_Name=@Item_Name, Item_Code=@Item_Code, HSN_Code=@HSN_Code, 
            ItemGroup_No=@ItemGroup_No, SubCategory_Sno=@SubCategory_Sno, Brand_Sno=@Brand_Sno, 
            Color_Sno=@Color_SNo, Weave_Sno=@Weave_Sno, ItemUnit_No=@ItemUnit_No, 
            AltUnit_No=@AltUnit_No, Conv_Factor=@Conv_Factor, GST_Sno=@GST_Sno, 
            Purchase_Rate=@Purchase_Rate, Selling_Price=@Selling_Price, IsActive=@IsActive, 
            Barcode=@Barcode, Item_Image=@Item_Image, Min_Stock=@Min_Stock, Max_Stock=@Max_Stock, 
            Batch_Enabled=@Batch_Enabled, Serial_Enabled=@Serial_Enabled, Modified_Date=GETDATE()
        WHERE Item_Sno = @Item_Sno AND Company_No = @Company_No;

        IF EXISTS (SELECT 1 FROM ClosingStk_table WHERE Item_no = @Item_Sno AND Company_No = @Company_No)
            UPDATE ClosingStk_table SET Stk_Value = @OpeningQty WHERE Item_no = @Item_Sno AND Company_No = @Company_No;
        ELSE
            INSERT INTO ClosingStk_table (Company_No, Item_no, Stk_Value, Stk_date) VALUES (@Company_No, @Item_Sno, @OpeningQty, GETDATE());
    END
END
