    Tables            : ItemGroup_Table
    Generated By      : Kavin

CREATE PROCEDURE sp_SaveCategory
    @Action            NVARCHAR(10),
    @ItemGroup_Sno     INT = 0,
    @Company_No        BIGINT,
    @User_No           INT,
    @ItemGroup_Name    NVARCHAR(100),
    @ItemGroup_Under   INT,
    @IsActive          BIT
AS
BEGIN
    SET NOCOUNT ON;

    -- Duplicate Check
    IF EXISTS (SELECT 1 FROM ItemGroup_Table WHERE Company_No = @Company_No AND ItemGroup_Name = @ItemGroup_Name AND ItemGroup_Sno <> @ItemGroup_Sno)
    BEGIN
        RAISERROR('This Category Name already exists.', 16, 1);
        RETURN;
    END

    IF @Action = 'INSERT'
    BEGIN
        INSERT INTO ItemGroup_Table (Company_No, User_No, ItemGroup_Name, ItemGroup_Under, IsActive, Created_Date)
        VALUES (@Company_No, @User_No, @ItemGroup_Name, @ItemGroup_Under, @IsActive, GETDATE());
    END
    ELSE
    BEGIN
        UPDATE ItemGroup_Table SET 
            ItemGroup_Name = @ItemGroup_Name, 
            ItemGroup_Under = @ItemGroup_Under, 
            IsActive = @IsActive 
        WHERE ItemGroup_Sno = @ItemGroup_Sno AND Company_No = @Company_No;
    END
END
GO
