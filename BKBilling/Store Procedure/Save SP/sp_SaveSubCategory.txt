    Description       : Save Sub-Category (Insert/Update)
    Tables            : ItemSubCategory_Table
    Generated By      : Kavin
    Version           : 1
*/

CREATE PROCEDURE sp_SaveSubCategory
    @Action         NVARCHAR(10),
    @SubCat_Sno     INT = 0,
    @Company_No     BIGINT,
    @User_No        INT,
    @Category_No    INT,
    @SubCat_Name    NVARCHAR(100),
    @IsActive       BIT
AS
BEGIN
    SET NOCOUNT ON;
    -- Duplicate Check within the same Category and Company
    IF EXISTS (SELECT 1 FROM ItemSubCategory_Table WHERE Company_No = @Company_No AND Category_No = @Category_No AND SubCat_Name = @SubCat_Name AND SubCat_Sno <> @SubCat_Sno)
    BEGIN
        RAISERROR('This Sub-Category already exists under the selected Category.', 16, 1);
        RETURN;
    END

    IF @Action = 'INSERT'
    BEGIN
        INSERT INTO ItemSubCategory_Table (Company_No, User_No, Category_No, SubCat_Name, IsActive)
        VALUES (@Company_No, @User_No, @Category_No, @SubCat_Name, @IsActive);
    END
    ELSE
    BEGIN
        UPDATE ItemSubCategory_Table SET 
            Category_No=@Category_No, SubCat_Name=@SubCat_Name, 
            IsActive=@IsActive, Modified_Date=GETDATE()
        WHERE SubCat_Sno = @SubCat_Sno AND Company_No = @Company_No;
    END
END
GO
