CREATE PROCEDURE sp_SaveUser
    @Action        NVARCHAR(10),
    @User_Sno      BIGINT = 0,
    @Company_No    BIGINT,
    @Username      NVARCHAR(50),
    @Password      NVARCHAR(MAX) = NULL,
    @FullName      NVARCHAR(100),
    @Role          NVARCHAR(50),
    @Phone         NVARCHAR(20),
    @Email         NVARCHAR(100),
    @Address_1     NVARCHAR(200),
    @Address_2     NVARCHAR(200),
    @IsActive      BIT,
    @Join_Date     DATE
AS
BEGIN
    SET NOCOUNT ON;

    -- Check for Duplicate Username within the same company
    IF EXISTS (SELECT 1 FROM User_Table WHERE Company_No = @Company_No AND Username = @Username AND User_Sno <> @User_Sno)
    BEGIN
        RAISERROR('The Username "%s" is already taken.', 16, 1, @Username);
        RETURN;
    END

    IF @Action = 'INSERT'
    BEGIN
        INSERT INTO User_Table (Company_No, Username, Password, FullName, Role, Phone, Email, Address_1, Address_2, IsActive, Join_Date, Created_Date)
        VALUES (@Company_No, @Username, @Password, @FullName, @Role, @Phone, @Email, @Address_1, @Address_2, @IsActive, @Join_Date, GETDATE());
    END
    ELSE IF @Action = 'UPDATE'
    BEGIN
        UPDATE User_Table SET 
            Username = @Username,
            FullName = @FullName,
            Role = @Role,
            Phone = @Phone,
            Email = @Email,
            Address_1 = @Address_1,
            Address_2 = @Address_2,
            IsActive = @IsActive,
            Join_Date = @Join_Date,
            Modified_Date = GETDATE(),
            Password = ISNULL(@Password, Password) -- Keep old password if new one is null
        WHERE User_Sno = @User_Sno;
    END
END
GO
