/*                                  
    Description       : Handles both Insertion and Updating of Area records with Duplicate Name validation.
    Tables            : Area_table
    Generated By      : Kavin
    Generated ON      : 07/07/2026
    Version           : 1
*/
CREATE PROCEDURE sp_SaveArea
    @Action NVARCHAR(10),
    @Area_Sno INT = 0,
    @Company_No BIGINT,
    @User_No INT,
    @Area_Name NVARCHAR(100),
    @Area_Under INT,
    @IsActive BIT
AS
BEGIN
    SET NOCOUNT ON;
    
    -- Check for duplicate Area Name in the same company
    IF EXISTS (SELECT 1 FROM Area_table WHERE Company_No = @Company_No AND Area_Name = @Area_Name AND Area_Sno <> @Area_Sno)
    BEGIN
        RAISERROR('This Area Name already exists for your company.', 16, 1);
        RETURN;
    END

    IF @Action = 'INSERT'
    BEGIN
        INSERT INTO Area_table (Company_No, User_No, Area_Name, Area_Under, IsActive, Created_Date)
        VALUES (@Company_No, @User_No, @Area_Name, @Area_Under, @IsActive, GETDATE());
    END
    ELSE IF @Action = 'UPDATE'
    BEGIN
        UPDATE Area_table SET 
            Area_Name = @Area_Name, 
            Area_Under = @Area_Under, 
            IsActive = @IsActive 
        WHERE Area_Sno = @Area_Sno AND Company_No = @Company_No;
    END
END
