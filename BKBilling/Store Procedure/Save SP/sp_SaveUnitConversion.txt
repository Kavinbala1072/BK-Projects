CREATE PROCEDURE sp_SaveUnitConversion
    @Action         NVARCHAR(10),
    @Conv_Sno       INT = 0,
    @Company_No     BIGINT,
    @MainUnit_Sno   INT,
    @SubUnit_Sno     INT,
    @Multiplier     DECIMAL(18, 4),
    @IsActive       BIT
AS
BEGIN
    SET NOCOUNT ON;

    -- Duplicate Check: Check if this specific unit pair already exists for this company
    IF EXISTS (SELECT 1 FROM UnitConversion_Table 
               WHERE Company_No = @Company_No 
               AND MainUnit_Sno = @MainUnit_Sno 
               AND SubUnit_Sno = @SubUnit_Sno 
               AND Conv_Sno <> @Conv_Sno)
    BEGIN
        RAISERROR('A conversion rule for this unit pair already exists.', 16, 1);
        RETURN;
    END

    IF @Action = 'INSERT'
    BEGIN
        INSERT INTO UnitConversion_Table (Company_No, MainUnit_Sno, SubUnit_Sno, Multiplier, IsActive, Created_Date)
        VALUES (@Company_No, @MainUnit_Sno, @SubUnit_Sno, @Multiplier, @IsActive, GETDATE());
    END
    ELSE IF @Action = 'UPDATE'
    BEGIN
        UPDATE UnitConversion_Table SET 
            MainUnit_Sno = @MainUnit_Sno, 
            SubUnit_Sno = @SubUnit_Sno, 
            Multiplier = @Multiplier, 
            IsActive = @IsActive 
        WHERE Conv_Sno = @Conv_Sno AND Company_No = @Company_No;
    END
END
