/*                                  
    Description       : Handles both the Creation (INSERT) and Modification (UPDATE) of 
                        ledger accounts. Includes logic to prevent duplicate Ledger Codes 
                        within the same company.
    Tables            : Ledger_Table
    Generated By      : Kavin
    Generated ON      : 07/07/2026
    Version           : 1
*/
CREATE PROCEDURE sp_SaveLedger
    @Action NVARCHAR(10),
    @Ledger_Sno BIGINT = 0,
    @Company_No BIGINT,
    @User_No INT,
    @ledger_Active BIT,
    @ledger_code NVARCHAR(50),
    @ledger_name NVARCHAR(150),
    @LedgerGroup_no INT,
    @Area_no INT,
    @ledger_Add1 NVARCHAR(200),
    @ledger_Add2 NVARCHAR(200),
    @ledger_Add3 NVARCHAR(200),
    @Ledger_Email NVARCHAR(100),
    @Ledger_Phone NVARCHAR(20),
    @Ledger_ContactPerson NVARCHAR(100),
    @ledger_bank NVARCHAR(100),
    @Ledger_Branch NVARCHAR(100),
    @ledger_AcNo NVARCHAR(50),
    @ledger_Ifscode NVARCHAR(20),
    @Ledger_PAN NVARCHAR(10),
    @Ledger_open DECIMAL(18,2),
    @Balance_Type NVARCHAR(10),
    @Credit_Limit DECIMAL(18,2),
    @Credit_Days INT,
    @Is_TDS_Applicable BIT,
    @Use_GST BIT,
    @Ledger_GST NVARCHAR(15),
    @GST_DealerType NVARCHAR(20),
    @GST_StateCode NVARCHAR(5),
    @Ledger_remarks NVARCHAR(MAX)
AS
BEGIN
    SET NOCOUNT ON;

    -- 1. Validation: Duplicate Code Check
    IF EXISTS (SELECT 1 FROM Ledger_Table WHERE Company_No = @Company_No AND ledger_code = @ledger_code AND Ledger_Sno <> @Ledger_Sno)
    BEGIN
        RAISERROR('The Ledger Code entered is already assigned to another account.', 16, 1);
        RETURN;
    END

    -- 2. Perform Action
    IF @Action = 'INSERT'
    BEGIN
        INSERT INTO Ledger_Table (
            Company_No, User_No, ledger_Active, ledger_code, ledger_name, LedgerGroup_no, Area_no,
            ledger_Add1, ledger_Add2, ledger_Add3, Ledger_Email, Ledger_Phone, Ledger_ContactPerson,
            ledger_bank, Ledger_Branch, ledger_AcNo, ledger_Ifscode, Ledger_PAN, Ledger_open,
            Balance_Type, Credit_Limit, Credit_Days, Is_TDS_Applicable, Use_GST, Ledger_GST,
            GST_DealerType, GST_StateCode, Ledger_remarks, Created_Date, Modified_Date
        ) VALUES (
            @Company_No, @User_No, @ledger_Active, @ledger_code, @ledger_name, @LedgerGroup_no, @Area_no,
            @ledger_Add1, @ledger_Add2, @ledger_Add3, @Ledger_Email, @Ledger_Phone, @Ledger_ContactPerson,
            @ledger_bank, @Ledger_Branch, @ledger_AcNo, @ledger_Ifscode, @Ledger_PAN, @Ledger_open,
            @Balance_Type, @Credit_Limit, @Credit_Days, @Is_TDS_Applicable, @Use_GST, @Ledger_GST,
            @GST_DealerType, @GST_StateCode, @Ledger_remarks, GETDATE(), GETDATE()
        );
    END
    ELSE
    BEGIN
        UPDATE Ledger_Table SET
            User_No = @User_No, 
            ledger_Active = @ledger_Active, 
            ledger_code = @ledger_code, 
            ledger_name = @ledger_name,
            LedgerGroup_no = @LedgerGroup_no, 
            Area_no = @Area_no, 
            ledger_Add1 = @ledger_Add1, 
            ledger_Add2 = @ledger_Add2,
            ledger_Add3 = @ledger_Add3, 
            Ledger_Email = @Ledger_Email, 
            Ledger_Phone = @Ledger_Phone,
            Ledger_ContactPerson = @Ledger_ContactPerson, 
            ledger_bank = @ledger_bank, 
            Ledger_Branch = @Ledger_Branch,
            ledger_AcNo = @ledger_AcNo, 
            ledger_Ifscode = @ledger_Ifscode, 
            Ledger_PAN = @Ledger_PAN,
            Ledger_open = @Ledger_open, 
            Balance_Type = @Balance_Type, 
            Credit_Limit = @Credit_Limit,
            Credit_Days = @Credit_Days, 
            Is_TDS_Applicable = @Is_TDS_Applicable, 
            Use_GST = @Use_GST,
            Ledger_GST = @Ledger_GST, 
            GST_DealerType = @GST_DealerType, 
            GST_StateCode = @GST_StateCode,
            Ledger_remarks = @Ledger_remarks, 
            Modified_Date = GETDATE()
        WHERE Ledger_Sno = @Ledger_Sno AND Company_No = @Company_No;
    END
END
GO
