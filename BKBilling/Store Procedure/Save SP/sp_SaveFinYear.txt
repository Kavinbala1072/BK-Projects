CREATE PROCEDURE sp_SaveFinYear
    @Company_No BIGINT,
    @FY_Name NVARCHAR(20),
    @StartDate DATE,
    @EndDate DATE
AS
BEGIN
    SET NOCOUNT ON;
    IF EXISTS (SELECT 1 FROM FinYear_Table WHERE Company_No = @Company_No AND FY_Name = @FY_Name)
    BEGIN
        RAISERROR('This Financial Year label already exists.', 16, 1);
        RETURN;
    END

    INSERT INTO FinYear_Table (Company_No, FY_Name, StartDate, EndDate)
    VALUES (@Company_No, @FY_Name, @StartDate, @EndDate);
END
GO
