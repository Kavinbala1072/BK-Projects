CREATE PROCEDURE sp_SaveColor
    @Action NVARCHAR(10),
    @Color_Sno INT = 0,
    @Company_No BIGINT,
    @User_No INT,
    @Color_Name NVARCHAR(100),
    @Color_HexCode NVARCHAR(20),
    @IsActive BIT
AS
BEGIN
    SET NOCOUNT ON;
    IF EXISTS (SELECT 1 FROM Color_Table WHERE Company_No = @Company_No AND Color_Name = @Color_Name AND Color_Sno <> @Color_Sno)
    BEGIN
        RAISERROR('This Color Name already exists.', 16, 1);
        RETURN;
    END

    IF @Action = 'INSERT'
    BEGIN
        INSERT INTO Color_Table (Company_No, User_No, Color_Name, Color_HexCode, IsActive)
        VALUES (@Company_No, @User_No, @Color_Name, @Color_HexCode, @IsActive);
    END
    ELSE
    BEGIN
        UPDATE Color_Table SET 
            Color_Name=@Color_Name, Color_HexCode=@Color_HexCode, 
            IsActive=@IsActive, Modified_Date=GETDATE()
        WHERE Color_Sno = @Color_Sno AND Company_No = @Company_No;
    END
END
GO
