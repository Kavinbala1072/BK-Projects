            window.onload = function () {
                // Monthly Data
                const sMonth = parseFloat(document.getElementById('<%= hfSalesMonth.ClientID %>').value) || 0;
                const pMonth = parseFloat(document.getElementById('<%= hfPurMonth.ClientID %>').value) || 0;
                // Today Data
                const sToday = parseFloat(document.getElementById('<%= hfSalesToday.ClientID %>').value) || 0;
                const pToday = parseFloat(document.getElementById('<%= hfPurToday.ClientID %>').value) || 0;
                // Outstanding Data
                const rec = parseFloat(document.getElementById('<%= hfRec.ClientID %>').value) || 0;
                const pay = parseFloat(document.getElementById('<%= hfPay.ClientID %>').value) || 0;

                // --- 1. NESTED TRADING CHART (Sales vs Purchase) ---
                new Chart(document.getElementById('chartTrading'), {
                    type: 'doughnut',
                    data: {
                        labels: ['Sales', 'Purchase'],
                        datasets: [
                            {
                                label: 'Monthly',
                                data: [sMonth, pMonth],
                                backgroundColor: ['#0d6efd', '#198754'], // Blue and Green
                                weight: 0.6
                            },
                            {
                                label: 'Today',
                                data: [sToday, pToday],
                                backgroundColor: ['#0d6efd', '#198754'], // Lighter shades for Today
                                weight: 0.4
                            }
                        ]
                    },
                    options: {
                        responsive: true,
                        maintainAspectRatio: false,
                        plugins: {
                            legend: { position: 'bottom' },
                            tooltip: {
                                callbacks: {
                                    label: function (context) {
                                        return context.dataset.label + " " + context.label + ": ₹" + context.raw.toLocaleString();
                                    }
                                }
                            }
                        }
                    }
                });

                // --- 2. OUTSTANDING CHART ---
                new Chart(document.getElementById('chartOut'), {
                    type: 'pie',
                    data: {
                        labels: ['Receivable (Dr)', 'Payable (Cr)'],
                        datasets: [{
                            data: [rec, pay],
                            backgroundColor: ['#0dcaf0', '#dc3545'],
                            hoverOffset: 10
                        }]
                    },
                    options: { responsive: true, maintainAspectRatio: false }
                });
            };
