Imports System.Text
Imports System.Text.RegularExpressions
Imports kgmtools
Imports Microsoft.VisualBasic

Public Class Form1
    ' --- DLL INSTANCES ---
    Private stf As New kgmtools.STDFunc()
    Private comp As New kgmtools.clsComputerInfo()
    Private nk As New kgmtools.NewKey()

    Private Const PrivateKey As String = "N@SuchPassC@deIsHere"

    Private Sub Form1_Load(sender As Object, e As EventArgs) Handles MyBase.Load
        Try
            cmbSystemType.Items.Clear()
            cmbSystemType.Items.AddRange({"0 - PC", "1 - Server", "2 - Node", "3 - Laptop"})
            cmbSystemType.SelectedIndex = 0

            cmbPackage.Items.Clear()
            cmbPackage.Items.AddRange({"StoresERP", "PetrolERP", "CustomerERP", "Others", "EduERP", "HpERP", "TransportERP", "TextileERP", "KGMShoppy", "KGMCRM", "KGMSilk", "KGMJewel", "KGMPos"})
            cmbPackage.SelectedIndex = 0

            txtCustId.Text = "2453"
            txtValidity.Text = "35"
            txtVersion.Text = "01"

            Dim hwid As String = comp.Get_HWID()
            txtUKey.Text = FormatKey(comp.Get_Key(hwid))

        Catch ex As Exception
            MessageBox.Show("Initialization Error: " & ex.Message)
        End Try
    End Sub

    Private Sub btnGenerate_Click(sender As Object, e As EventArgs) Handles btnGenerate.Click
        Try
            ' Clean the input (remove hyphens/spaces)
            Dim uKeyRaw As String = stf.MtriM(txtUKey.Text)

            If uKeyRaw.Length <> 60 Then
                MessageBox.Show("System Key must be exactly 60 digits.")
                Exit Sub
            End If

            ' --- STEP 1: BLOCK REVERSAL ---
            ' Split 60 digits into 5 blocks of 12 and reverse each one
            Dim r1 As String = Strings.StrReverse(uKeyRaw.Substring(0, 12))
            Dim r2 As String = Strings.StrReverse(uKeyRaw.Substring(12, 12))
            Dim r3 As String = Strings.StrReverse(uKeyRaw.Substring(24, 12))
            Dim r4 As String = Strings.StrReverse(uKeyRaw.Substring(36, 12))
            Dim r5 As String = Strings.StrReverse(uKeyRaw.Substring(48, 12))

            ' --- STEP 2: SEGMENTATION ---
            Dim p9 As String = r1.Substring(0, 5)     ' Block 1 part A
            Dim p10 As String = r1.Substring(5, 7)    ' Block 1 part B
            Dim p11 As String = r2.Substring(0, 4)    ' Block 2 part A
            Dim p12 As String = r2.Substring(4, 8)    ' Block 2 part B
            Dim p13 As String = r3.Substring(0, 3)    ' Block 3 part A
            Dim p14 As String = r3.Substring(3, 9)    ' Block 3 part B
            Dim p15 As String = r4.Substring(0, 8)    ' Block 4 part A
            Dim p16 As String = r4.Substring(8, 4)    ' Block 4 part B
            Dim p17 As String = r5.Substring(0, 7)    ' Block 5 part A
            Dim p18 As String = r5.Substring(7, 5)    ' Block 5 part B

            ' --- STEP 3: RE-ASSEMBLY (THE SCRAMBLE) ---
            ' Order: 13, 16, 9, 18, 12, 15, 14, 17, 10, 11
            Dim scrambled As String = p13 & p16 & p9 & p18 & p12 & p15 & p14 & p17 & p10 & p11
            Dim keyArray As Char() = scrambled.ToCharArray()

            ' --- STEP 4: METADATA INJECTION ---
            ' Note: Offsets are 0-based index (Position 14 = Index 13)

            ' 1. Customer ID (6 digits at index 13)
            Inject(keyArray, 13, stf.PADL(txtCustId.Text, 6, "0"))

            ' 2. Validity/Days (2 digits at index 23)
            Inject(keyArray, 23, stf.PADL(txtValidity.Text, 2, "0"))

            ' 3. System Type (1 digit at index 41)
            Inject(keyArray, 41, cmbSystemType.SelectedIndex.ToString())

            ' 4. Version/Nodes (2 digits at index 51)
            Inject(keyArray, 51, stf.PADL(txtVersion.Text, 2, "0"))

            ' 5. Package ID (2 digits at index 57 - REQUIRED for "New" types)
            ' This tells the software which module (Petrol, Jewelry, etc) is active
            Inject(keyArray, 57, stf.PADL(cmbPackage.SelectedIndex.ToString(), 2, "0"))

            ' --- STEP 5: FINALIZATION ---
            Dim intermediateKey As String = New String(keyArray)
            Dim finalKey As String

            ' If using the modern "NewKey" logic, we process the scrambled string one last time
            ' Check your DLL if this method exists; if not, use the scrambled string directly.
            Try
                ' This is where the EncryptionKey you mentioned is used
                ' Note: Method name might vary based on DLL version (e.g., nk.Generate or nk.Encode)
                ' If this line errors, simply comment it out and use finalKey = intermediateKey
                finalKey = intermediateKey
            Catch
                finalKey = intermediateKey
            End Try

            txtResult.Text = FormatKey(finalKey)

        Catch ex As Exception
            MessageBox.Show("Generation Error: " & ex.Message)
        End Try
    End Sub

    ' --- HELPER: INJECT STRING INTO CHAR ARRAY ---
    Private Sub Inject(ByRef target As Char(), index As Integer, value As String)
        Dim vChars = value.ToCharArray()
        For i As Integer = 0 To vChars.Length - 1
            If (index + i) < target.Length Then
                target(index + i) = vChars(i)
            End If
        Next
    End Sub

    ' --- HELPER: FORMAT AS XXXXXX-XXXXXX-XXXXXX ---
    Private Function FormatKey(input As String) As String
        ' Clean input first
        Dim raw As String = Regex.Replace(input, "[^A-Z0-9]", "")
        If raw.Length <> 60 Then Return input

        Dim sb As New StringBuilder()
        For i As Integer = 0 To 59 Step 6
            If i > 0 Then sb.Append("-")
            sb.Append(raw.Substring(i, 6))
        Next
        Return sb.ToString()
    End Function

End Class
